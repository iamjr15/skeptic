import { test } from "skeptic-cli";

test("jigyansurout — B7-B10 verification: cursor narration, --video-size, audit.md, daemon", async ({
  page,
  observability,
  screenshot,
}) => {
  await page.goto("https://jigyansurout.com/");
  await page.waitForLoadState("domcontentloaded");
  await page.waitForTimeout(800);
  await page.mouse.move(640, 360);
  await page.waitForTimeout(300);
  await page.mouse.move(420, 240);
  await page.waitForTimeout(500);
  await screenshot("01-home");

  await page.goto("https://jigyansurout.com/projects");
  await page.waitForLoadState("domcontentloaded");
  await page.waitForTimeout(800);
  await page.mouse.move(500, 300);
  await page.waitForTimeout(300);
  await page.mouse.move(720, 420);
  await page.waitForTimeout(500);
  await screenshot("02-projects");

  await page.goto("https://jigyansurout.com/achievements");
  await page.waitForLoadState("domcontentloaded");
  await page.waitForTimeout(800);
  await page.mouse.move(640, 360);
  await page.waitForTimeout(300);
  await page.mouse.move(380, 540);
  await page.waitForTimeout(500);
  await screenshot("03-achievements");

  await page.goto("https://jigyansurout.com/blog");
  await page.waitForLoadState("domcontentloaded");
  await page.waitForTimeout(800);
  await page.mouse.move(640, 360);
  await page.waitForTimeout(300);
  await page.mouse.move(800, 280);
  await page.waitForTimeout(500);
  await screenshot("04-blog");

  // B9 — dual-engine audit. impacts:["critical"] suppresses fail since the
  // page has known non-critical violations; the audit still runs and the
  // full violation list lands in audit.md.
  try {
    await observability.expectAccessible({ standard: "WCAG21AA", impacts: ["critical"] });
  } catch {
    // page has known non-critical violations
  }
});
